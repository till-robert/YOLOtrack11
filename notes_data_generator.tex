    \section*{Bead reference stack}

    \begin{itemize}
        \item Original images: monochrome $16 \,\text{bit}$ tiff, $\qtyproduct{512x512}{\pixel}$, $\qty{0.161}{\um\per\pixel}$
        \item The center is at $(x_0,y_0)=(\qty{259}{\pixel},\qty{256}{\pixel})$
        \item Number of $z$ slices: $1568$, $\qty{0.134}{\um}$ spacing
        \item $z=0$ is chosen to be at the focus
        \item[$\rightarrow$] \textbf{Field of view} $(x_{\text{FOV}},y_{\text{FOV}},z_{\text{FOV}})=(82.4,82.4,210)\,\unit{\um}$
        \item Images are ...
        \begin{itemize}
            \item radially averaged
            \item Downsampled by a factor of $\alpha=1/4$
        \end{itemize} 
        \item[$\rightarrow$]  Final (numpy) shape: $\numproduct{1568x128x128}$, with scale $\qty{0.644}{\um\per\pixel}$
    \end{itemize}
    \section*{Training data generator}
        \begin{itemize}
            \item Bounding box width/height (set arbitrarily by eye)
            \begin{equation*}
                bw = bh = \left(\frac{0.21\cdot|z|}{\qty{0.161}{\um}}+55\right)  \,\unit{\pixel}
            \end{equation*}
            \item (Downsampled) particles are placed on a background, with intensity chosen randomly between $\qty{90}{\percent}$ and $\qty{110}{\percent}$ of original intensity
            \item The background FOV is determined by \begin{equation*}
                \text{background FOV}=\frac{\text{backgr. imgsz.}}{\text{refstack imgsz.}}\cdot \text{refstack FOV}
            \end{equation*}
            \item we choose...
            \begin{itemize}
                {\color{gray} \item  either $\qtyproduct{640x540}{\pixel}$, corresponding to a FOV of $(x_{\text{FOV}},y_{\text{FOV}})=(412, 348) \, \upmu\text{m}$}
                \item or (closer to Yundon's FOV) $\qtyproduct{646x545}{\pixel}$, corresponding to a FOV of $(x_{\text{FOV}},y_{\text{FOV}})=(416,351) \, \upmu\text{m}$, with a subsequent downscale to $\qtyproduct{640x540}{\pixel}$
            \end{itemize}
            \item this allows to use of input data of the format of Yundon's stacks ($\qtyproduct{1280x1080}{\pixel}$) by simply downscaling the images by a factor of $1/2$.
        \item Gaussian noise with $\sigma^2 \approx \numrange{45}{50}$ is added on top of the images, the same amount as Yundon's images when downscaled
        \item $z$-position is normalized to lie between $0$ and $1$ and denoted by $\tilde{z}$,
        \begin{equation*}
            \tilde{z} = \frac{z}{\qty{400}{\um}}+\num{0.5}
            \Leftrightarrow z = (\tilde{z} - \num{0.5}) \cdot \qty{400}{\um},
        \end{equation*}
        thus

    \section*{Yundon's bacteria stack}
    \begin{itemize}
    \item For Yundon's images (which have a spacing of $\qty{0.2676}{\um}$), we need to calculate the $z$-position of each particle relative to the slice number $i$ from their absolute $z$-positions $z_{\mathrm{abs.}}$:
        \begin{equation*}
            z_i = z_{\mathrm{abs.}} - \qty{0.2676}{\um} \cdot i
        \end{equation*}
        A tracked particle is discarded if its $z_i$ is below $-\qty{102}{\um}$ or above $\qty{108}{\um}$ as the model cannot output $z$-values outside this interval
    \end{itemize}
